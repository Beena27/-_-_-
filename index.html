        // Share result

        function shareResult(platform) {

            const studentName = document.getElementById('studentFullName').textContent;

            const totalScore = document.getElementById('totalScore').textContent;

           

            let shareUrl = '';

            let message = `የ${studentName} የመሠረተ ሃይማኖት ፈተና ውጤት: ${totalScore}/60`;

           

            if (platform === 'whatsapp') {

                shareUrl = `https://wa.me/?text=${encodeURIComponent(message)}`;

            } else if (platform === 'telegram') {

                shareUrl = `https://t.me/share/url?url=${encodeURIComponent(window.location.href)}&text=${encodeURIComponent(message)}`;

            }

           

            if (shareUrl) {

                window.open(shareUrl, '_blank');

            }

        }
